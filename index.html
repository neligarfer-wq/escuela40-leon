function buscarRobot(texto){

 

texto=texto.toLowerCase();

 

let robots=document.querySelectorAll(".robot");

 

robots.forEach(r=>{

 

if(r.innerText.toLowerCase().includes(texto))

r.classList.add("show");

 

else

r.classList.remove("show");

 

});

 

}
